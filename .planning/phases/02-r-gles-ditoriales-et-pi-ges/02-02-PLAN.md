// src/lib/content-core/rules/types.ts — types + one pure helper, no node: import
import type { Document, LineCounter } from 'yaml';
export type Mode = 'preview' | 'production';
export type Severite = 'error' | 'warning';
export interface Finding {
  code: string;            // must be in CATALOGUE
  severity?: Severite;     // default 'error'
  pointer: string;         // JSON pointer in `file` ('' = root)
  message: string;         // French; quote contributor text only through citer()
  file?: string;           // absolute path; default = the entry / file under examination
  at?: { line: number; col: number }; // explicit position (non-YAML files such as ids.lock, or a held AST position)
}
export interface LoadedEntry { collection: CollectionName; id: string; file: string; data: Record<string, unknown>; doc: Document.Parsed; lc: LineCounter }
export interface FichierAnalyse { collection: CollectionName; id: string; file: string; doc: Document.Parsed; lc: LineCounter }
export interface Corpus {
  entries: readonly LoadedEntry[];                                   // schema-valid + path-coherent entries
  de(collection: CollectionName): readonly LoadedEntry[];
  entree(collection: CollectionName, id: string): LoadedEntry | undefined;
  listes: ReadonlyMap<CollectionName, ReadonlySet<string>>;          // ids of ALL listed files, valid or not (no cascade)
  aDesInvalides(collection: CollectionName): boolean;                // a listed file of this collection is not in entries
}
export interface RuleContext { mode: Mode; contentRoot: string; corpus: Corpus; cache: Map<string, unknown> }
export interface FileRule { id: string; run(f: FichierAnalyse, ctx: RuleContext): Finding[] }
export interface EntryRule { id: string; run(e: LoadedEntry, ctx: RuleContext): Finding[] }
export interface CorpusRule { id: string; run(ctx: RuleContext): Finding[] }
export interface Famille { id: string; fichier: readonly FileRule[]; entree: readonly EntryRule[]; corpus: readonly CorpusRule[] }
export function citer(s: string, max?: number): string; // cleans whitespace, truncates to CITER_MAX (60) chars with « … »

// src/lib/content-core/rules/index.ts
export const FAMILLES: readonly Famille[]; // [redaction, corpus, sondages] in this order (02-06 appends citations)
export function appliquerRegles(args: {
  mode: Mode; contentRoot: string; familles: readonly Famille[];
  fichiers: readonly FichierAnalyse[];      // every file whose YAML was parsed, even if its schema failed
  entries: readonly LoadedEntry[];          // valid entries
  listes: ReadonlyMap<CollectionName, ReadonlySet<string>>;
}): Diagnostic[];

// src/lib/content-core/rules/{redaction,corpus,sondages}.ts — registration modules, one per wave-2 plan
export const famille: Famille; // created here as { id, fichier: [], entree: [], corpus: [] }; the owning plan fills the arrays

// src/lib/content-core/load.ts
export interface LoadOptions { schemasDir: string; mode?: Mode; familles?: readonly Famille[] } // familles defaults to FAMILLES (tests inject synthetic ones)
